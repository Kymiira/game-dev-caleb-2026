Demonstrating continuous keyframe rotation combined with requestAnimationFrame translation script:

/* css.js movement loop snippet */
const img = document.getElementById('img');
let direction = 1;
let position = 0;
function move() {
    position += direction * 3;
    img.style.left = position + 'px';
    if (position >= 100) {
        direction = -1;
    } else if (position <= -100) {
        direction = 1;
    } 
    requestAnimationFrame(move);
}
